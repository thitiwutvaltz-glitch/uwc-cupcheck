function rxAddMed_(u, req){

  var storage = req.storage==='หัวเตียง' ? 'หัวเตียง' : 'ตลับจัดยา';
  if (!name) throw new Error('กรุณาใส่ชื่อยา');
  if (!unit) throw new Error('กรุณาเลือกหน่วย');
  if (!(dose > 0 && dose <= 20)) throw new Error('จำนวนต่อครั้งไม่ถูกต้อง');
  if (qty < 0 || qty > 5000) throw new Error('จำนวนที่รับไม่ถูกต้อง');
  var photos = (req.photos||[]).filter(function(f){ return /^Receive(Batch|Photo)_Images\/[\w.]+\.jpg$/.test(f); });
  if (qty > 0 && !photos.length) throw new Error('ยังไม่ได้ถ่ายรูปยา');

  try{
    var res = table_('Residents').rows.filter(function(r){ return r.ResidentID===req.residentId; })[0];
    if (!res || res.Branch!==BRANCH) throw new Error('ไม่พบคนไข้ในสาขานี้');

      Note:'เพิ่มผ่าน LINE โดย ' + who, Active:true, Storage:storage, Instruction:String(req.instruction||'').trim().slice(0,200), LowFlag:false,
      InitQty:qty || '', InitExpiry:req.expiry||'', BatchID:batch, MealTiming:'', Frequency:freq, DoseQty:dose, FreqTime:/^(06|10|22):00$/.test(String(req.freqTime||'')) ? String(req.freqTime) : '', InitPhoto:batchPhoto}]);
    rxAppend_('ReceiveBatch', [{BatchID:batch, DateTime:now, ResidentID:res.ResidentID, Branch:BRANCH, StaffName:who, Note:'เพิ่มยาใหม่ผ่าน LINE' + (req.note ? ' · ' + String(req.note).slice(0,150) : ''), Photo:batchPhoto}]);
    if (qty > 0) rxAppend_('Transactions', [{TxnID:rxId_(), DateTime:now, MedID:medId, ResidentID:res.ResidentID, Branch:BRANCH, Type:'รับเข้า', Qty:qty, ExpectedQty:0, Diff:0, RecountStatus:'',
      UnitPrice:0, Amount:0, ExpiryDate:req.expiry||'', Photo:'', DoseChanged:false, StaffName:who, CentralTxnID:'', Note:'รับตอนเพิ่มยาใหม่ (LINE)', BatchID:batch}]);

    SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
  var sync = {}; try{ CacheService.getScriptCache().remove('cupSync'); sync = cupSync_(); }catch(e){ sync = {ok:false}; }
  return {ok:true, medId:medId, sync:sync};
}
function rxStopMed_(u, req){
  var reason = String(req.reason||'').trim().slice(0,150);
  if (!reason) throw new Error('กรุณาใส่เหตุผล');
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try{
    var tm = table_('Medications'), m = tm.rows.filter(function(r){ return r.MedID===req.medId; })[0];
    if (!m) throw new Error('ไม่พบยานี้');
    var res = table_('Residents').rows.filter(function(r){ return r.ResidentID===m.ResidentID; })[0];
    if (!res || res.Branch!==BRANCH) throw new Error('ยานี้ไม่ใช่ของสาขานี้');
    if (String(m.Active).toUpperCase()!=='TRUE') throw new Error('ยานี้หยุดไปแล้ว');
    var sd = m.StartDate instanceof Date ? m.StartDate : new Date(m.StartDate);
    if (isNaN(sd) || Date.now()-sd.getTime() > RX_CANCEL_DAYS*86400000) throw new Error('หยุดยาจากหน้านี้ได้เฉพาะยาที่เพิ่มไม่เกิน ' + RX_CANCEL_DAYS + ' วัน ยาเก่าให้หยุดในแอประบบยา');
    var who = u.StaffID || u.StaffName, stamp = who + ' ' + now_() + ': ' + reason;
    writeCells_(tm, m._row, {Active:false, StopDate:Utilities.formatDate(new Date(), TZ, 'MM/dd/yyyy'), Note:(m.Note ? m.Note + ' · ' : '') + 'หยุด (เพิ่มผิด) โดย ' + stamp});
    var tt = table_('Transactions'), nc = 0;
    tt.rows.forEach(function(t){ if (t.MedID===m.MedID && ((t.Type==='รับเข้า' && String(t.Note).indexOf('เพิ่มยาใหม่')>=0) || (t.Type==='ปรับยอด' && String(t.Note).indexOf('ย้ายยอดจากยาเดิม')>=0))){ writeCells_(tt, t._row, {Type:RX_CANCEL_TYPE, Note:(t.Note ? t.Note + ' · ' : '') + 'ยกเลิก (ยาเพิ่มผิด) โดย ' + stamp}); nc++; } });
    var back = (String(m.Note).match(/เปลี่ยนจาก (M\w+)/)||[])[1], restored = '';
    if (back){ var om = tm.rows.filter(function(r){ return r.MedID===back; })[0];
      if (om && String(om.Active).toUpperCase()!=='TRUE'){ writeCells_(tm, om._row, {Active:true, StopDate:'', Note:(om.Note ? om.Note + ' · ' : '') + 'กลับมาใช้ (ยกเลิกการเปลี่ยนวิธีกิน) โดย ' + stamp}); restored = back; } }
    var cupsN = rxCupRemove_(m.MedID);
    SpreadsheetApp.flush();
    var out0 = {ok:true, cancelledReceipts:nc, cupsUpdated:cupsN, restored:restored};
  } finally { lock.releaseLock(); }
  if (out0.restored){ try{ CacheService.getScriptCache().remove('cupSync'); out0.sync = cupSync_(); }catch(e){} }
  return out0;
}

function rxNewMeds_(residentId){
  var since = Date.now() - RX_CANCEL_DAYS*86400000;
  return table_('Medications').rows.filter(function(m){
    if (m.ResidentID!==residentId || String(m.Active).toUpperCase()!=='TRUE' || !/(เพิ่ม|เปลี่ยนวิธีกิน)ผ่าน LINE/.test(String(m.Note))) return false;
    var d = m.StartDate instanceof Date ? m.StartDate : new Date(m.StartDate); return !isNaN(d) && d.getTime() >= since - 86400000;
  }).map(function(m){ return {medId:m.MedID, name:String(m.DrugName||'')+(m.Strength?' }

/* ---------- หมอเปลี่ยนวิธีกิน/ขนาด: หยุดยาเดิม + สร้างยาใหม่ + ย้ายยอดคงเหลือ (ปรับยอด) + อัปเดตถ้วยวันนี้ ---------- */
function rxCupRemove_(medId){
  var pre = Utilities.formatDate(new Date(), TZ, 'yyyyMMdd') + '-', tc = table_('Med_Cup'), wait = {};
  tc.rows.forEach(function(c){ if (String(c.CupID).indexOf(pre)===0 && c.Status==='รอจัด') wait[c.CupID] = c; });
  var ti = table_('Med_Cup_Item'), del = [], left = {}, touched = {};
  ti.rows.forEach(function(i){ if (!wait[i.CupID]) return; if (i.MedID===medId){ del.push(i._row); touched[i.CupID] = 1; } else left[i.CupID] = (left[i.CupID]||0) + (Number(i.PrepQty)||0); });
  del.sort(function(a,b){ return b-a; }).forEach(function(r){ ti.sh.deleteRow(r); });
  var emptyRows = []; Object.keys(touched).forEach(function(id){ var c = wait[id]; if (left[id]) writeCells_(tc, c._row, {PrepTotal:left[id]}); else emptyRows.push(c._row); });
  emptyRows.sort(function(a,b){ return b-a; }).forEach(function(r){ tc.sh.deleteRow(r); });
  return Object.keys(touched).length;
}

      MealTiming:'', Frequency:freq, DoseQty:dose, FreqTime:ftime, Note:'เปลี่ยนวิธีกินผ่าน LINE โดย ' + who + ' · เปลี่ยนจาก ' + m.MedID + (why ? ' · ' + why : '')});
    delete nm._row;
    writeCells_(tm, m._row, {Active:false, StopDate:today, Note:(m.Note ? m.Note + ' · ' : '') + 'หยุด (หมอเปลี่ยนวิธีกิน/ขนาด → ' + newId + ') โดย ' + who + ' ' + now + (why ? ': ' + why : '')});
    rxAppend_('Medications', [nm]);
    if (carry > 0) rxAppend_('Transactions', [{TxnID:rxId_(), DateTime:now, MedID:newId, ResidentID:m.ResidentID, Branch:BRANCH, Type:'ปรับยอด', Qty:carry, ExpectedQty:0, Diff:0, RecountStatus:'',
      UnitPrice:0, Amount:0, ExpiryDate:'', Photo:'', DoseChanged:true, StaffName:who, CentralTxnID:'', Note:'ย้ายยอดจากยาเดิม ' + m.MedID + ' (หมอเปลี่ยนวิธีกิน/ขนาด · LINE)', BatchID:''}]);
    var cups = rxCupRemove_(m.MedID);
    SpreadsheetApp.flush();
    out = {ok:true, medId:newId, oldId:m.MedID, carry:carry, cupsRemoved:cups};
  } finally { lock.releaseLock(); }
  try{ CacheService.getScriptCache().remove('cupSync'); out.sync = cupSync_(); }catch(e){ out.sync = {ok:false}; }
  return out;
}
